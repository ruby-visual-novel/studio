function changeThemeBrightness(val) { const body = document.getElementById('mainBody'); const header = document.getElementById('mainHeader'); const sidebar = document.getElementById('sidebar'); const v = parseInt(val); // ค้นหาการ์ดและกล่องข้อความทั้งหมดเพื่อปรับสีตัวหนังสือและขอบให้สอดคล้องกับความสว่าง const cards = document.querySelectorAll('.theme-card'); const inputs = document.querySelectorAll('input, select, textarea'); if (v === 0) { // โหมดมืด (Dark Mode ปกติ) body.style.backgroundColor = '#000000'; body.style.color = '#ffffff'; header.style.backgroundColor = '#18181b'; header.style.borderColor = '#27272a'; if(sidebar) sidebar.style.backgroundColor = '#18181b'; cards.forEach(card => { card.style.backgroundColor = '#18181b'; // bg-zinc-900 card.style.borderColor = '#27272a'; card.style.color = '#ffffff'; }); } else { // โหมดสว่าง / ปรับพื้นหลังขาว const rgbVal = Math.round(255 * (v / 100)); const bgHex = `rgb(${rgbVal}, ${rgbVal}, ${rgbVal})`; if (v > 50) { // พื้นหลังค่อนข้างสว่างมาก -> ตัวหนังสือและกรอบต้องเป็นสีเข้ม (ดำ/เทาเข้ม) body.style.backgroundColor = bgHex; body.style.color = '#111111'; header.style.backgroundColor = `rgb(${Math.max(0, rgbVal-20)}, ${Math.max(0, rgbVal-20)}, ${Math.max(0, rgbVal-20)})`; header.style.borderColor = '#d1d5db'; if(sidebar) sidebar.style.backgroundColor = bgHex; cards.forEach(card => { card.style.backgroundColor = `rgb(${Math.max(220, rgbVal-10)}, ${Math.max(220, rgbVal-10)}, ${Math.max(220, rgbVal-10)})`; card.style.borderColor = '#cbd5e1'; card.style.color = '#111111'; }); } else { // พื้นหลังโทนสว่างปานกลาง body.style.backgroundColor = bgHex; body.style.color = '#f3f4f6'; header.style.backgroundColor = `rgb(${Math.min(255, rgbVal+15)}, ${Math.min(255, rgbVal+15)}, ${Math.min(255, rgbVal+15)})`; header.style.borderColor = '#4b5563'; if(sidebar) sidebar.style.backgroundColor = bgHex; cards.forEach(card => { card.style.backgroundColor = `rgb(${Math.min(240, rgbVal+10)}, ${Math.min(240, rgbVal+10)}, ${Math.min(240, rgbVal+10)})`; card.style.borderColor = '#4b5563'; card.style.color = '#f3f4f6'; }); } } }
